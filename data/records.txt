const DATA_FILES = {
  records: 'data/records.txt',
  papers: 'data/papers.txt',
  partners: 'data/partners.txt',
};

async function loadTextFile(filePath) {
  const response = await fetch(filePath, { cache: 'no-store' });
  if (!response.ok) {
    throw new Error(`${filePath}를 불러오지 못했습니다.`);
  }
  return response.text();
}

function parseRecordLine(line) {
  const cleanLine = line.trim();
  if (!cleanLine) return null;

  const match = cleanLine.match(/^([0-9]+):\{([^:]+):(\d+),(\d+),(\d+):(\d+)\}$/);
  if (!match) return null;

  const [, examNumber, name, background, operation, military, round] = match;

  return {
    examNumber,
    name,
    background: Number(background),
    operation: Number(operation),
    military: Number(military),
    round: Number(round),
  };
}

function parseLinkLine(line) {
  const cleanLine = line.trim();
  if (!cleanLine) return null;

  const separatorIndex = cleanLine.indexOf(' - ');
  if (separatorIndex === -1) return null;

  const title = cleanLine.slice(0, separatorIndex).trim();
  const url = cleanLine.slice(separatorIndex + 3).trim();

  return { title, url };
}

function renderRecordResult(record) {
  const resultBox = document.getElementById('record-result');
  if (!record) {
    resultBox.textContent = '조회 결과가 없습니다.';
    resultBox.classList.add('empty');
    return;
  }

  resultBox.classList.remove('empty');
  resultBox.innerHTML = `
    <div class="record-card">
      <div class="record-header">
        <h3>${record.name}</h3>
        <span class="exam-badge">수험번호 ${record.examNumber}</span>
      </div>
      <ul class="score-list">
        <li>배경지식: ${record.background}점</li>
        <li>운영: ${record.operation}점</li>
        <li>군사: ${record.military}점</li>
      </ul>
      <p class="round-note">제 ${record.round}회 VNAT 시험 성적</p>
    </div>
  `;
}

async function loadRecordData() {
  const text = await loadTextFile(DATA_FILES.records);
  const lines = text.split(/\r?\n/);
  const records = lines
    .map(parseRecordLine)
    .filter(Boolean);

  return records;
}

async function loadListData(filePath) {
  const text = await loadTextFile(filePath);
  const lines = text.split(/\r?\n/);
  return lines
    .map(parseLinkLine)
    .filter(Boolean);
}

async function renderLinkList(listId, filePath) {
  const listEl = document.getElementById(listId);
  try {
    const items = await loadListData(filePath);
    if (!items.length) {
      listEl.innerHTML = '<li>등록된 항목이 없습니다.</li>';
      return;
    }

    listEl.innerHTML = items
      .map(
        (item) => `
          <li>
            <a href="${item.url}" target="_blank" rel="noopener noreferrer">${item.title}</a>
          </li>
        `
      )
      .join('');
  } catch (error) {
    listEl.innerHTML = '<li>자료를 불러오지 못했습니다.</li>';
  }
}

async function initializeRecords() {
  try {
    const records = await loadRecordData();

    document.getElementById('search-record').addEventListener('click', () => {
      const query = document.getElementById('exam-no').value.trim();
      const matchedRecord = records.find((record) => record.examNumber === query);
      renderRecordResult(matchedRecord || null);
    });

    if (records.length > 0) {
      renderRecordResult(records[0]);
    }
  } catch (error) {
    renderRecordResult(null);
  }
}

async function initializeLists() {
  await Promise.all([
    renderLinkList('papers-list', DATA_FILES.papers),
    renderLinkList('partners-list', DATA_FILES.partners),
  ]);
}

document.addEventListener('DOMContentLoaded', async () => {
  await initializeRecords();
  await initializeLists();
});
